cmake_minimum_required(VERSION 3.20)
project(MarketReplay VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(SANITIZER "none" CACHE STRING "none, address, or thread")
find_package(Threads REQUIRED)
add_library(market src/book.cpp src/parser.cpp src/replay.cpp)
target_include_directories(market PUBLIC include)
target_link_libraries(market PUBLIC Threads::Threads)
target_compile_options(market PUBLIC -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
if(SANITIZER STREQUAL "address")
  target_compile_options(market PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(market PUBLIC -fsanitize=address,undefined)
elseif(SANITIZER STREQUAL "thread")
  target_compile_options(market PUBLIC -fsanitize=thread -fno-omit-frame-pointer)
  target_link_options(market PUBLIC -fsanitize=thread)
elseif(NOT SANITIZER STREQUAL "none")
  message(FATAL_ERROR "Unknown SANITIZER=${SANITIZER}")
endif()
add_executable(replay src/main.cpp)
target_link_libraries(replay PRIVATE market)
include(CTest)
if(BUILD_TESTING)
  add_executable(unit_tests tests/unit.cpp)
  target_link_libraries(unit_tests PRIVATE market)
  add_test(NAME unit COMMAND unit_tests)
  set_tests_properties(unit PROPERTIES TIMEOUT 30)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME differential COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/differential.py $<TARGET_FILE:replay>)
  set_tests_properties(differential PROPERTIES TIMEOUT 120)
endif()
